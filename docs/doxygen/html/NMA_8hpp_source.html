107inline void jacobiEigh(std::vector<Real> A, int n, std::vector<Real>& evalOut, std::vector<Real>& evecOut) {
108 evecOut.assign(static_cast<std::size_t>(n) * n, Real(0));
109 for (int i = 0; i < n; ++i) {
110 evecOut[i * n + i] = Real(1);
111 }
112
113 const int maxSweeps = 100;
114 for (int sweep = 0; sweep < maxSweeps; ++sweep) {
115 Real off = 0;
116 for (int p = 0; p < n; ++p) {
117 for (int q = p + 1; q < n; ++q) {
118 off += A[p * n + q] * A[p * n + q];
119 }
120 }
121 if (off <= Real(1e-30)) {
122 break;
123 }
124
125 for (int p = 0; p < n; ++p) {
126 for (int q = p + 1; q < n; ++q) {
127 const Real apq = A[p * n + q];
128 if (std::abs(apq) < Real(1e-300)) {
129 continue;
130 }
131 const Real app = A[p * n + p];
132 const Real aqq = A[q * n + q];
133 const Real phi = Real(0.5) * std::atan2(Real(2) * apq, aqq - app);
134 const Real c = std::cos(phi), srot = std::sin(phi);
135 for (int k = 0; k < n; ++k) {
136 const Real akp = A[k * n + p], akq = A[k * n + q];
137 A[k * n + p] = c * akp - srot * akq;
138 A[k * n + q] = srot * akp + c * akq;
139 }
140 for (int k = 0; k < n; ++k) {
141 const Real apk = A[p * n + k], aqk = A[q * n + k];
142 A[p * n + k] = c * apk - srot * aqk;
143 A[q * n + k] = srot * apk + c * aqk;
144 }
145 for (int k = 0; k < n; ++k) {
146 const Real vkp = evecOut[k * n + p], vkq = evecOut[k * n + q];
147 evecOut[k * n + p] = c * vkp - srot * vkq;
148 evecOut[k * n + q] = srot * vkp + c * vkq;
149 }
150 }
151 }
152 }
153
154 std::vector<int> ord(static_cast<std::size_t>(n));
155 for (int i = 0; i < n; ++i) {
156 ord[i] = i;
157 }
158 std::vector<Real> diag(static_cast<std::size_t>(n));
159 for (int i = 0; i < n; ++i) {
160 diag[i] = A[i * n + i];
161 }
162 std::stable_sort(ord.begin(), ord.end(), [&](int x, int y) {
163 return diag[x] < diag[y];
164 });
165
166 evalOut.assign(static_cast<std::size_t>(n), Real(0));
167 std::vector<Real> evecSorted(static_cast<std::size_t>(n) * n, Real(0));
168 for (int newc = 0; newc < n; ++newc) {
169 const int oldc = ord[newc];
170 evalOut[newc] = diag[oldc];
171 for (int r = 0; r < n; ++r) {
172 evecSorted[r * n + newc] = evecOut[r * n + oldc];
173 }
174 }
175 evecOut.swap(evecSorted);
176}

206inline RouteBNMA
207computeRouteBNMA(const RobotModel& model, RobotState& state, Real h = Real(1e-5), Real zeroTol = Real(1e-6)) {
208 const int nu = model.nu;
209 RouteBNMA out;
210 out.nu = nu;
211 if (nu == 0) {
212 return out;
213 }
214
215 // snapshot the minimized generalized coordinates
216 std::vector<Real> q0(static_cast<std::size_t>(model.nq));
217 {
218 const Real* q = state.q();
219 for (int i = 0; i < model.nq; ++i) {
220 q0[i] = q[i];
221 }
222 }
223 auto restoreQ0 = [&]() {
224 Real* q = state.q();
225 for (int i = 0; i < model.nq; ++i) {
226 q[i] = q0[i];
227 }
228 };
229
230 // scratch
231 std::vector<OpenMM::Vec3> posScratch, forceScratch;
232 std::vector<Vec3> atomForce;
233 std::vector<Real> dvel(static_cast<std::size_t>(nu), Real(0));
234 std::vector<Real> gPlus, gMinus;
235
236 std::cout << "[RouteBNMA] computing mass-weighted internal-coordinate Hessian at q0..." << std::endl;
237
238 // ---- 1. generalized stiffness K (column-by-column central difference) ----
239 // K stored row-major nu x nu; column b is the response to perturbing s_b.
240 std::vector<Real> K(static_cast<std::size_t>(nu) * nu, Real(0));
241 for (int b = 0; b < nu; ++b) {
242 restoreQ0();
243 dvel[b] = +h;
244 ConstraintSet::applyVelSpaceIncrementToQ(model, state, dvel.data());
245 RobotEngine::realizePosition(model, state);
246 detail::generalizedForce(model, state, posScratch, forceScratch, atomForce, gPlus);
247
248 restoreQ0();
249 dvel[b] = -h;
250 ConstraintSet::applyVelSpaceIncrementToQ(model, state, dvel.data());
251 RobotEngine::realizePosition(model, state);
252 detail::generalizedForce(model, state, posScratch, forceScratch, atomForce, gMinus);
253
254 dvel[b] = Real(0);
255 const Real inv2h = Real(1) / (Real(2) * h);
256 for (int a = 0; a < nu; ++a) {
257 // g = -dV/ds => K_ab = d2V/ds_a ds_b = -(gPlus - gMinus)/2h
258 K[a * nu + b] = -(gPlus[a] - gMinus[a]) * inv2h;
259 }
260 }
261 std::cout << "[RouteBNMA] ...done." << std::endl;
262
263 // symmetrize (kills the antisymmetric part from numerical force noise)
264 for (int i = 0; i < nu; ++i) {
265 for (int j = i + 1; j < nu; ++j) {
266 const Real avg = Real(0.5) * (K[i * nu + j] + K[j * nu + i]);
267 K[i * nu + j] = avg;
268 K[j * nu + i] = avg;
269 }
270 }
271
272 std::cout << "[RouteBNMA] diagonalizing Htilde = N^T K N..." << std::endl;
273
274 // ---- 2. mass-weighting factor N = sqrt(M^-1) (N N^T = M^-1) -------------
275 // Build N at the minimized configuration. N column j = sqrtMInv(e_j).
276 restoreQ0();
277 RobotEngine::realizePosition(model, state);
278 RobotEngine::realizeArticulatedBodyInertias(model, state); // fills P, G, DI for sqrtMInv
279
280 std::vector<Real> N(static_cast<std::size_t>(nu) * nu, Real(0));
281 {
282 std::vector<Real> ej(static_cast<std::size_t>(nu), Real(0));
283 std::vector<Real> col(static_cast<std::size_t>(nu), Real(0));
284 for (int j = 0; j < nu; ++j) {
285 std::fill(ej.begin(), ej.end(), Real(0));
286 ej[j] = Real(1);
287 // NB: multiplyBySqrtMInv uses V_GB as scratch -- harmless here.
288 RobotEngine::multiplyBySqrtMInv(model, state, ej.data(), col.data());
289 for (int i = 0; i < nu; ++i) {
290 N[i * nu + j] = col[i];
291 }
292 }
293 }
294
295 std::cout << "[RouteBNMA] ...done." << std::endl;
296
297 // ---- 3. Htilde = N^T K N -------------------------------------------------
298 std::vector<Real> KN(static_cast<std::size_t>(nu) * nu, Real(0)); // K @ N
299 for (int i = 0; i < nu; ++i) {
300 for (int j = 0; j < nu; ++j) {
301 Real acc = 0;
302 for (int k = 0; k < nu; ++k) {
303 acc += K[i * nu + k] * N[k * nu + j];
304 }
305 KN[i * nu + j] = acc;
306 }
307 }
308 std::vector<Real> Htil(static_cast<std::size_t>(nu) * nu, Real(0)); // N^T @ KN
309 for (int i = 0; i < nu; ++i) {
310 for (int j = 0; j < nu; ++j) {
311 Real acc = 0;
312 for (int k = 0; k < nu; ++k) {
313 acc += N[k * nu + i] * KN[k * nu + j];
314 }
315 Htil[i * nu + j] = acc;
316 }
317 }
318 for (int i = 0; i < nu; ++i) { // re-symmetrize against roundoff
319 for (int j = i + 1; j < nu; ++j) {
320 const Real avg = Real(0.5) * (Htil[i * nu + j] + Htil[j * nu + i]);
321 Htil[i * nu + j] = avg;
322 Htil[j * nu + i] = avg;
323 }
324 }
325
326 std::cout << "[RouteBNMA] ...done." << std::endl;
327
328 // ---- 4. diagonalize ------------------------------------------------------
329 std::vector<Real> eval, evec;
330 detail::jacobiEigh(Htil, nu, eval, evec); // eval ascending; evec col k = y_k
331 out.eigval = eval;
332
333 // u-space modes v_k = N y_k
334 out.modeU.assign(static_cast<std::size_t>(nu), std::vector<Real>(static_cast<std::size_t>(nu), Real(0)));
335 for (int k = 0; k < nu; ++k) {
336 for (int i = 0; i < nu; ++i) {
337 Real acc = 0;
338 for (int j = 0; j < nu; ++j) {
339 acc += N[i * nu + j] * evec[j * nu + k];
340 }
341 out.modeU[k][i] = acc;
342 }
343 }
344
345 std::cout << "[RouteBNMA] ...done." << std::endl;
346
347 // count near-zero (rigid/external/noise) modes and pick the softest above tol
348 out.nNearZero = 0;
349 for (int k = 0; k < nu; ++k) {
350 if (std::abs(eval[k]) <= zeroTol) {
351 ++out.nNearZero;
352 }
353 }
354 out.softMode = -1;
355 for (int k = 0; k < nu; ++k) {
356 if (eval[k] > zeroTol) {
357 out.softMode = k;
358 break;
359 }
360 }
361 if (out.softMode < 0) {
362 out.softMode = nu - 1; // degenerate fallback
363 }
364
365 // ---- 5. hand-off vector for the NMA path (unit direction in u-space) -----
366 out.uScaleFactors = out.modeU[static_cast<std::size_t>(out.softMode)];
367 Real nrm = 0;
368 for (Real x : out.uScaleFactors) {
369 nrm += x * x;
370 }
371 nrm = std::sqrt(nrm);
372 if (nrm > Real(0)) {
373 for (Real& x : out.uScaleFactors) {
374 x /= nrm;
375 }
376 }
377
378 restoreQ0();
379 RobotEngine::realizePosition(model, state); // leave caller's state at q0
380 return out;
381}
