80inline void driftPositions(const RobotModel& m,
81 RobotState& s,
82 const std::vector<Real>& q0,
83 const std::vector<Real>& qdot0,
84 const std::vector<Real>& u0,
85 const std::vector<Real>& udot0,
86 const std::vector<Real>& qdd0,
87 Real h,
88 int nq) {
89 Real* q = s.q();
90 for (int i = 0; i < nq; ++i) {
91 q[i] = q0[i] + (h * qdot0[i]) + ((h * h / 2) * qdd0[i]);
92 }
93
94 for (int bodyIx = 1; bodyIx < m.numBodies; ++bodyIx) {
95 if (!m.isQuaternionBody(bodyIx)) {
96 continue;
97 }
98 const int qOff = m.bodyQIndex[bodyIx];
99 const int uOff = m.bodyUIndex[bodyIx];
100 std::array<Real, 4> qNew;
101 robo::jointDriftQuat(
102 m.bodyJoint[bodyIx], q0.data(), qOff, u0.data(), udot0.data(), uOff, s.X_FM()[bodyIx], h, qNew.data());
103 q[qOff + 0] = qNew[0];
104 q[qOff + 1] = qNew[1];
105 q[qOff + 2] = qNew[2];
106 q[qOff + 3] = qNew[3];
107 // translation block (Free q[qOff+4..6], FreeLine q[qOff+4..6]) keeps its
108 // Taylor update from the loop above; Ball has no translation block.
109 }
110 RobotEngine::normalizeQuaternions(m, s); // mops up ~1e-16 rounding after the exp-map; never rescues an overflow
111}

200template <class VelSaneFn, class EvalVelFn, class RestoreFn>
201inline bool velocityCorrector(const std::vector<Real>& u0,
202 const std::vector<Real>& udot0,
203 Real h,
204 int nu,
205 Real* u,
206 const Real* udot,
207 VelSaneFn&& velocitiesSane,
208 EvalVelFn&& evalVel,
209 RestoreFn&& restorePreStep,
210 bool* correctorConverged) {
211 for (int i = 0; i < nu; ++i) {
212 u[i] = u0[i] + (h * udot0[i]); // u1_est
213 }
214
215 // second half of the original combined `evalPos() || evalVel()` check;
216 // the caller already verified evalPos() before calling this function.
217 if (!evalVel()) {
218 restorePreStep();
219 return false;
220 }
221
222 // Simbody: tol = min(1e-4, 0.1*accuracy). For fixed-step HMC (default accuracy
223 // ~1e-3) this evaluates to 1e-4. Plain functional iteration, no under-relaxation,
224 // max 10 sweeps -- matching VerletIntegrator::attemptDAEStep exactly.
225 const Real tol = Real(1e-4);
226 Real prevChange = std::numeric_limits<Real>::infinity();
227 Real lastChange = std::numeric_limits<Real>::infinity(); // for the dt-too-large message
228 int usedIters = 0;
229 bool converged = false;
230
231 for (int iter = 0; iter < 10; ++iter) {
232 ++usedIters;
233 Real num = 0;
234 Real den = 0; // Simbody's relative 2-norm change
235
236 for (int i = 0; i < nu; ++i) {
237 const Real un = u0[i] + ((h / 2) * (udot0[i] + udot[i]));
238 const Real d = un - u[i];
239 num += d * d;
240 den += u[i] * u[i];
241 u[i] = un;
242 }
243
244 // Genuine non-finite / runaway. In Simbody this is the realize()/project()
245 // exception path: caught, and in fixed-step mode the step still "succeeds",
246 // propagating the bad state to the move-level energy validation, which
247 // rejects the MOVE. We short-circuit to the same outcome by rejecting here.
248 if (!velocitiesSane()) {
249 restorePreStep();
250 return false;
251 }
252 // q is unchanged by the corrector, so positions/forces (evalPos) are already
253 // current from the single evaluation above; only re-derive velocity terms.
254 if (!evalVel()) {
255 restorePreStep();
256 return false;
257 }
258
259 const Real change = std::sqrt(num) / (std::sqrt(den) + Real(1e-30));
260 lastChange = change;
261 if (!std::isfinite(change) || change > Real(1e6)) {
262 restorePreStep();
263 return false;
264 }
265
266 if (change <= tol) {
267 converged = true;
268 break; // converged
269 }
270
271 // Functional iteration stopped contracting (after iter > 1, to skip the
272 // crude forward-Euler seed's first non-monotone blip). We stop iterating
273 // here; whether to take the step or reject is decided after the loop.
274 if (iter > 1 && change > prevChange) {
275 break;
276 }
277
278 prevChange = change;
279 }
280
281 // dt-too-large guard (deliberately STRICTER than Simbody's "take the step").
282 // Reaching here without convergence means the implicit-trapezoid corrector
283 // could not find its fixed point at this dt for the CURRENT configuration --
284 // a FINITE, bounded solve that simply will not contract. This is distinct from
285 // a steric clash (non-finite force / runaway u), which is handled above as a
286 // move rejection (return false) because clashes are a normal, transient part of
287 // sampling.
288 //
289 // THEORY (boilerplate "Propagation" / RATTLE): the fixed step is taken
290 // UNCONDITIONALLY -- "a non-converged corrector does not shrink dt or reject
291 // the step ... the best available velocity estimate is accepted", and "it is
292 // the trajectory-level Metropolis test, not per-step control, that supplies
293 // correctness." So on non-convergence we KEEP the advanced position q (set by
294 // the position drift before this loop; the corrector only refines u) and the
295 // last/best velocity iterate, and return success. A step that pumps energy
296 // then shows up as a large dH and is rejected at the MOVE level (the caller
297 // restores its saved q), which is the correct, theory-sanctioned behavior.
298 //
299 // The previous code instead called restorePreStep() here and still returned
300 // true: that silently UNDID the step (q == q0) while reporting success, so the
301 // move-level dH was always ~0 and every proposal was "accepted" as a no-op --
302 // i.e. the trajectory froze in place. That both violated the theory above and
303 // produced exactly that frozen-coordinate symptom; it is removed. The warning
304 // is kept (fail-loud diagnostic): a non-converged corrector means dt is too
305 // large for this geometry, so most such steps will be rejected by Metropolis
306 // until the world's timestep is reduced.
307 if (correctorConverged) {
308 *correctorConverged = converged;
309 }
310 if (!converged) {
311 std::fprintf(stderr,
312 "[verlet] world: velocity corrector did not converge at dt=%.6g ps "
313 "(relative change %.3e > tol %.3e after %d iterations). The timestep is "
314 "likely too large for the current configuration; the step is still taken "
315 "(THEORY: trajectory-level Metropolis supplies correctness) but expect "
316 "rejections until the timestep is reduced.\n",
317 (double)h,
318 (double)lastChange,
319 (double)tol,
320 usedIters);
321 }
322 return true;
323}

325template <class Bridge>
326bool RobotEngine::verletStep(const RobotModel& m,
327 RobotState& s,
328 Bridge& bridge,
329 const robo::ConstraintSet& cset,
330 Real h,
331 bool* correctorConverged) {
332 if (correctorConverged) {
333 *correctorConverged = false; // pessimistic default; set true only on a converged step below
334 }
335 const int nq = m.nq, nu = m.nu;
336 Real* q = s.q();
337 Real* u = s.u();
338 Real* qdot = s.qdot();
339 Real* udot = s.udot();
340 Real* qdd = s.qdotdot();
341
342 std::vector<Real> q0(q, q + nq);
343 std::vector<Real> u0(u, u + nu);
344 std::vector<Real> qdot0(qdot, qdot + nq);
345 std::vector<Real> udot0(udot, udot + nu);
346 std::vector<Real> qdd0(qdd, qdd + nq);
347
348 // ---- Cartesian-integrated solvent (solvent-relaxing NCMC) ----------------
349 // A subset of atoms is advanced in FLAT Cartesian space by velocity-Verlet
350 // driven by the same OpenMM force evaluation that drives the solute internal
351 // step -- so the contact environment relaxes INSIDE the proposal instead of
352 // being a welded wall (docs/specs/ncmc_solvent_relax.md). This shares the
353 // existing integrator's structure exactly: position drift x1 = x0 + h v0 +
354 // (h^2/2) a0, then the velocity half is the same implicit trapezoid the
355 // generalized speeds use -- but the Cartesian force depends only on position
356 // (no velocity-dependent terms), so that trapezoid is EXPLICIT and needs no
357 // iteration: v1 = v0 + (h/2)(a0 + a1). The map is the standard symmetric,
358 // time-reversible, volume-preserving Verlet, so the move's exact-dH HMC
359 // acceptance argument is preserved (checkReversibility certifies it). The
360 // block is skipped entirely when no atom is flagged (the welded engine),
361 // making every existing path bit-identical.
362 const std::vector<int>& solvAtoms = s.cartSolventAtoms();
363 const std::vector<Real>& solvInvM = s.cartSolventInvMass();
364 const int nSolv = static_cast<int>(solvAtoms.size());
365 Vec3* posG = s.atomPosG();
366 Vec3* velG = s.atomVelG();
367 Vec3* frcG = s.atomForceG();
368 std::vector<Vec3> xs0(nSolv), vs0(nSolv), fs0(nSolv);
369 for (int j = 0; j < nSolv; ++j) {
370 const int a = solvAtoms[j];
371 xs0[j] = posG[a]; // x0
372 vs0[j] = velG[a]; // v0
373 fs0[j] = frcG[a]; // a0 = f0 * invMass (cached by the previous force eval)
374 }
375
376 // True iff every per-body spatial force is finite. OpenMM returns Inf forces
377 // for a hard steric overlap (LJ r^-12); integrating against those is what
378 // produced the "Particle coordinate is NaN" crash. Catch it here, at the
379 // source, and signal the caller to reject this step's pose.
380 auto forcesFinite = [&]() -> bool {
381 const SpatialVec* bf = s.bodyForceG();
382 for (int b = 1; b < m.numBodies; ++b) {
383 if (!std::isfinite(bf[b][0][0]) || !std::isfinite(bf[b][0][1]) || !std::isfinite(bf[b][0][2])
384 || !std::isfinite(bf[b][1][0]) || !std::isfinite(bf[b][1][1])
385 || !std::isfinite(bf[b][1][2])) {
386 return false;
387 }
388 }
389 return true;
390 };
391
392 // forcesFinite() only catches OpenMM Inf. A CFL-unstable step (e.g. a stiff
393 // explicit-solvent contact at too-large dt) amplifies u geometrically while
394 // it is still finite, then overflows the quaternion. Cap ||u|| relative to
395 // the freshly-seeded momentum norm so a runaway is rejected early, while a
396 // merely hot trajectory (||u|| growing a few x) passes.
397 Real uSeedNorm2 = 0;
398 for (int i = 0; i < nu; ++i) {
399 uSeedNorm2 += u0[i] * u0[i];
400 }
401 const Real uCap2 = (uSeedNorm2 + Real(1e-30)) * Real(1e6);
402 auto velocitiesSane = [&]() -> bool {
403 Real n2 = 0;
404 for (int i = 0; i < nu; ++i) {
405 if (!std::isfinite(u[i])) {
406 return false;
407 }
408 n2 += u[i] * u[i];
409 }
410 return n2 <= uCap2;
411 };
412
413 auto restorePreStep = [&]() {
414 std::copy(q0.begin(), q0.end(), q);
415 std::copy(u0.begin(), u0.end(), u);
416 std::copy(qdot0.begin(), qdot0.end(), qdot);
417 std::copy(udot0.begin(), udot0.end(), udot);
418 std::copy(qdd0.begin(), qdd0.end(), qdd);
419 realizePosition(m, s);
420 fillAtomPositionsFromBodies(m, s);
421 // Cartesian solvent is owned by posG/velG, not by any body, so it is not
422 // rolled back by the fill above -- restore it explicitly.
423 for (int j = 0; j < nSolv; ++j) {
424 const int a = solvAtoms[j];
425 posG[a] = xs0[j];
426 velG[a] = vs0[j];
427 frcG[a] = fs0[j];
428 }
429 };
430
431 // ---- position drift ----
432 driftPositions(m, s, q0, qdot0, u0, udot0, qdd0, h, nq);
433
434 // Cartesian solvent position drift: x1 = x0 + h v0 + (h^2/2) a0, a0 = f0/m.
435 // Written straight into posG; fillAtomPositionsFromBodies (below) skips these
436 // atoms, so the drift survives and feeds the OpenMM force eval.
437 cartesianSolventDrift(solvAtoms, solvInvM, xs0, vs0, fs0, posG, h, nSolv);
438
439 auto refreshPos = [&]() {
440 realizePosition(m, s);
441 fillAtomPositionsFromBodies(m, s);
442 };
443 refreshPos();
444 cset.enforcePositionConstraints(m, s, refreshPos); // localProjectQ
445
446 // Position/force derivatives (evalPos) are a pure function of q, and the velocity
447 // corrector below NEVER changes q -- only u. So realizePosition and the OpenMM force
448 // evaluation (the dominant per-step cost) are identical across all corrector sweeps
449 // and are HOISTED to run exactly once here, not once per sweep. Only the
450 // velocity-dependent work (evalVel: realizeVelocity, the ABA inertias' velocity-
451 // coupled centrifugal seed, calcUDot, qdots) iterates. This is bitwise-identical to
452 // the old per-sweep evalDerivs (same q -> same forces every sweep) but evaluates
453 // OpenMM once/step instead of up to 11x/step.
454 auto evalPos = [&]() -> bool {
455 realizePosition(m, s);
456 ROBO_CHECK("realizePosition");
457 bridge.evaluate(s);
458 ROBO_CHECK("bridge.evaluate");
459 if (!forcesFinite()) {
460 return false; // non-finite force -> abort before it corrupts udot/q
461 }
462 // The articulated-inertia factorization (P/PPlus/D/DI/G) is a pure function of q,
463 // so hoist it here alongside realizePosition; the corrector below re-derives only
464 // the velocity-dependent seed (evalVel), turning the per-body Jacobi eigensolves
465 // from ~11x/step into 1x/step. Bitwise-identical (same q -> same factorization
466 // every sweep). See docs/specs/gpu-cartesian-kinematics/03-aba-parallelization Sec.0.5.
467 factorizeArticulatedInertias(m, s);
468 return true;
469 };
470 auto evalVel = [&]() -> bool {
471 realizeVelocity(m, s);
472 ROBO_CHECK("realizeVelocity");
473 seedArticulatedCentrifugal(m, s); // abcf = P*a_mob + gyro (P from the hoisted factorization)
474 calcUDot(m, s);
475 ROBO_CHECK("calcUDot");
476 for (int i = 0; i < nu; ++i) {
477 if (!std::isfinite(udot[i])) {
478 return false; // finite-but-diverging udot -> reject before it propagates
479 }
480 }
481 calcQDot(m, s, qdot);
482 calcQDotDot(m, s);
483 return true;
484 };
485 // evalPos-only half of the original combined `evalPos() || evalVel()` check;
486 // evalVel's half now runs as velocityCorrector's own first action below, so
487 // the short-circuit (evalVel only runs if evalPos succeeded) is preserved.
488 if (!evalPos()) {
489 restorePreStep();
490 return false;
491 }
492
493 // ---- velocity: implicit trapezoid + functional iteration ----
494 if (!velocityCorrector(u0, udot0, h, nu, u, udot, velocitiesSane, evalVel, restorePreStep, correctorConverged)) {
495 return false;
496 }
497
498 cset.enforceVelocityConstraints(m, s); // localProjectU (RATTLE)
499 realizeVelocity(m, s); // refresh V/KE at the projected u
500
501 // Cartesian solvent velocity half: v1 = v0 + (h/2)(a0 + a1), a1 = f1/m. The
502 // positions are frozen through the solute corrector, so f1 = frcG (evaluated
503 // once at the drifted x1 by evalPos) is the end-of-step force -- the update
504 // is explicit and exact, no iteration. This completes the symmetric Verlet.
505 cartesianSolventKick(solvAtoms, solvInvM, vs0, fs0, frcG, velG, h, nSolv);
506
507 s.time += h;
508 return true;
509}

511template <class Bridge>
512auto RobotEngine::stepTo(const RobotModel& model,
513 RobotState& state,
514 Bridge& bridge,
515 const robo::ConstraintSet& cset,
516 Real tEnd,
517 bool* correctorConverged) -> bool {
518 const Real h = tEnd - state.time;
519 if (h <= 0) {
520 if (correctorConverged) {
521 *correctorConverged = true; // no-op step: trivially "converged"
522 }
523 return true;
524 }
525 return verletStep(model, state, bridge, cset, h, correctorConverged);
526}

528template <class Bridge>
529// ============================================================================
530// Reversibility diagnostic (HMC proposal sanity check)
531// ============================================================================
532// Integrate nSteps forward at fixed step h, flip every generalized speed
533// (u -> -u), integrate nSteps "back", flip again. For a time-reversible map the
534// state returns to its start to within ~machine epsilon amplified by the work
535// done; a too-large h (the non-reversible, energy-pumTorsiong regime) returns a
536// residual of order the trajectory size. Returns the RELATIVE round-trip
537// residual ||(q,u)_returned - (q,u)_start|| / ||(q,u)_start||.
538//
539// Two properties of this probe matter for how it is used:
540// * It is NON-DESTRUCTIVE: the state s is restored to its entry value before
541// returning, so it can be called at startup or mid-run without perturbing
542// the chain.
543// * It certifies h ONLY for the configuration it is run from. The integrator
544// map depends on q through the configuration-dependent mass metric M(q) and
545// the local force stiffness, so the largest reversible h varies across
546// configuration space (THEORY 5.5). A startup call is therefore a smoke test
547// -- "is dt in the right ballpark for this geometry?" -- NOT a whole-run
548// guarantee. The ongoing, per-configuration guard is the corrector-
549// convergence throw inside verletStep, which tests the CURRENT geometry on
550// every step.
551//
552// If the corrector throws mid-probe (dt already non-convergent here), that is
553// caught and reported as an infinite residual rather than aborting the probe.
554//
555// Quaternion DOF are compared with a double-cover-aware chordal distance
556// min(|q - q0|, |q + q0|), since q and -q are the same rotation on S^3 (THEORY
557// 3.4); scalar coordinates and the velocity vector use the plain Euclidean norm.
558Real RobotEngine::checkReversibility(const RobotModel& m,
559 RobotState& s,
560 Bridge& bridge,
561 const robo::ConstraintSet& cset,
562 int nSteps,
563 Real h) {
564 const int nq = m.nq, nu = m.nu;
565
566 // Snapshot the entry state (for comparison and for restoration).
567 std::vector<Real> qStart(s.q(), s.q() + nq);
568 std::vector<Real> uStart(s.u(), s.u() + nu);
569 std::vector<Real> qdotStart(s.qdot(), s.qdot() + nq);
570 std::vector<Real> udotStart(s.udot(), s.udot() + nu);
571 std::vector<Real> qddStart(s.qdotdot(), s.qdotdot() + nq);
572 const Real timeStart = s.time;
573
574 // Cartesian solvent (solvent-relaxing NCMC): the joint map (q,u,x_s,v_s) must
575 // be momentum-flip reversible too, so the probe carries the solvent DOF.
576 const std::vector<int>& solvAtoms = s.cartSolventAtoms();
577 const int nSolv = static_cast<int>(solvAtoms.size());
578 Vec3* posG = s.atomPosG();
579 Vec3* velG = s.atomVelG();
580 std::vector<Vec3> xsStart(nSolv), vsStart(nSolv);
581 for (int j = 0; j < nSolv; ++j) {
582 xsStart[j] = posG[solvAtoms[j]];
583 vsStart[j] = velG[solvAtoms[j]];
584 }
585
586 auto restoreStart = [&]() {
587 std::copy(qStart.begin(), qStart.end(), s.q());
588 std::copy(uStart.begin(), uStart.end(), s.u());
589 std::copy(qdotStart.begin(), qdotStart.end(), s.qdot());
590 std::copy(udotStart.begin(), udotStart.end(), s.udot());
591 std::copy(qddStart.begin(), qddStart.end(), s.qdotdot());
592 s.time = timeStart;
593 realizePosition(m, s);
594 realizeVelocity(m, s);
595 fillAtomPositionsFromBodies(m, s);
596 for (int j = 0; j < nSolv; ++j) {
597 posG[solvAtoms[j]] = xsStart[j];
598 velG[solvAtoms[j]] = vsStart[j];
599 }
600 };
601
602 // Momentum-flip involution S: negate all generalized speeds and refresh the
603 // velocity-dependent derived quantities against -u. Cartesian solvent speeds
604 // are flipped in lockstep (the joint involution F).
605 auto flipMomenta = [&]() {
606 Real* u = s.u();
607 for (int i = 0; i < nu; ++i) {
608 u[i] = -u[i];
609 }
610 for (int j = 0; j < nSolv; ++j) {
611 Vec3& v = velG[solvAtoms[j]];
612 v = Vec3(-v[0], -v[1], -v[2]);
613 }
614 realizeVelocity(m, s);
615 realizeArticulatedBodyInertias(m, s);
616 calcUDot(m, s);
617 calcQDot(m, s, s.qdot());
618 calcQDotDot(m, s);
619 };
620
621 // Forward leg, flip, back leg, flip. Any non-finite rejection or corrector
622 // throw means h is already not integrable/reversible here -> infinite residual.
623 try {
624 for (int i = 0; i < nSteps; ++i) {
625 if (!verletStep(m, s, bridge, cset, h)) {
626 restoreStart();
627 return std::numeric_limits<Real>::infinity();
628 }
629 }
630 flipMomenta();
631 for (int i = 0; i < nSteps; ++i) {
632 if (!verletStep(m, s, bridge, cset, h)) {
633 restoreStart();
634 return std::numeric_limits<Real>::infinity();
635 }
636 }
637 flipMomenta();
638 } catch (const std::exception&) {
639 restoreStart();
640 return std::numeric_limits<Real>::infinity();
641 }
642
643 // Round-trip residual. Quaternion blocks: double-cover chordal distance.
644 const Real* q = s.q();
645 const Real* u = s.u();
646 Real resid2 = 0, scale2 = 0;
647
648 std::vector<bool> isQuatSlot(nq, false);
649 for (int b = 1; b < m.numBodies; ++b) {
650 if (!m.isQuaternionBody(b)) {
651 continue;
652 }
653 const int qOff = m.bodyQIndex[b];
654 Real dPlus2 = 0, dMinus2 = 0;
655 for (int k = 0; k < 4; ++k) {
656 isQuatSlot[qOff + k] = true;
657 const Real a = q[qOff + k], b0 = qStart[qOff + k];
658 dPlus2 += (a - b0) * (a - b0);
659 dMinus2 += (a + b0) * (a + b0);
660 scale2 += b0 * b0;
661 }
662 resid2 += std::min(dPlus2, dMinus2); // q == -q on S^3
663 }
664 for (int i = 0; i < nq; ++i) {
665 if (isQuatSlot[i]) {
666 continue;
667 }
668 const Real d = q[i] - qStart[i];
669 resid2 += d * d;
670 scale2 += qStart[i] * qStart[i];
671 }
672 for (int i = 0; i < nu; ++i) {
673 const Real d = u[i] - uStart[i];
674 resid2 += d * d;
675 scale2 += uStart[i] * uStart[i];
676 }
677 // Cartesian solvent positions and velocities (plain Euclidean -- flat space).
678 for (int j = 0; j < nSolv; ++j) {
679 const Vec3& x = posG[solvAtoms[j]];
680 const Vec3& v = velG[solvAtoms[j]];
681 for (int k = 0; k < 3; ++k) {
682 const Real dx = x[k] - xsStart[j][k];
683 const Real dv = v[k] - vsStart[j][k];
684 resid2 += dx * dx + dv * dv;
685 scale2 += xsStart[j][k] * xsStart[j][k] + vsStart[j][k] * vsStart[j][k];
686 }
687 }
688
689 restoreStart(); // non-destructive probe
690 return std::sqrt(resid2) / (std::sqrt(scale2) + Real(1e-30));
691}
