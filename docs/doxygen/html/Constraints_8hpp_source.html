134 std::vector<Real>(numU, 0));

136 std::vector<Vec3> atomForce(static_cast<std::size_t>(model.numAtoms), Vec3(0));
137
138 int iter = 0;
139 for (; iter < maxIter; ++iter) {
140 refreshKinematics();
141
142 Real worst = 0;
143 for (int con = 0; con < numC; ++con) {
144 const DistanceConstraint& bond = distance[con];
145 const Vec3 vecAB = assembleConstraintRow(model, state, bond, atomForce,
146 jacobianT[con].data(), minvJacobianT[con].data());
147 violation[con] = dot(vecAB, vecAB) - (bond.restLength * bond.restLength);
148 worst = std::max(worst, std::abs(violation[con]));
149 }
150 if (worst <= tolerance) {
151 break;
152 }
153
154 std::vector<Real> halfViolation(static_cast<std::size_t>(numC), 0);
155 for (int con = 0; con < numC; ++con) {
156 halfViolation[con] = Real(0.5) * violation[con]; // dC/dq = 2 G
157 }
158 std::vector<Real> lambda = solveCoupling(jacobianT, minvJacobianT, halfViolation, numC, numU);
159
160 std::vector<Real> deltaVel(static_cast<std::size_t>(numU), 0);
161 for (int con = 0; con < numC; ++con) {
162 for (int idx = 0; idx < numU; ++idx) {
163 deltaVel[idx] -= lambda[con] * minvJacobianT[con][idx];
164 }
165 }
166 applyVelSpaceIncrementToQ(model, state, deltaVel.data());
167 }
168 return iter;
169 }
