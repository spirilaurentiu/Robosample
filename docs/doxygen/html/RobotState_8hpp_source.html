3// ============================================================================
4// RobotState - the single aligned slab holding every per-step quantity, SoA.
5// Replaces SimTK::State + the SBTree*Cache caches (Simbody01/Simbody/src/
6// SimbodyTreeState.h). Field names track those caches as provenance only;
7// Simbody has been removed from the build, so there is no longer a live
8// 1:1 diff against it (see the VALIDATION note in RobotEngine.cpp for the
9// oracle-of-record that replaced it).
10//
11// H_FM / H / G are stored as SpatialVec COLUMNS: column j of a body lives at
12// index (bodyUIndex + j). One SpatialVec == 2 Vec3, so an array of `nu`
13// SpatialVec is exactly the 2*nu Vec3 hinge storage Simbody uses.
14//
15// DI (the dof x dof articulated hinge inverse) lives in a flat Real pool of
16// size model.nuSq, addressed by model.bodyUSqIndex[b] (row-major dof x dof).
17// ============================================================================

519 // ---- Cartesian-integrated solvent set (per-world, runtime) ---------------
520 // The atoms whose flat-space Cartesian (x,v) are advanced by velocity-Verlet
521 // inside the proposal (solvent-relaxing NCMC). Empty => the world behaves
522 // exactly as the welded torsional engine (every integrator/energy path below
523 // is guarded on emptiness), which is what keeps the existing tests bit-exact.
524 // cartSolventAtoms_ : the atom indices (drives the O(n_solv) Verlet loops)
525 // cartSolventInvMass_ : 1/m for each, parallel to cartSolventAtoms_
526 // cartSolventMask_ : per-atom 0/1 for O(1) skip in fillAtomPositionsFromBodies
540 void setCartSolvent(const std::vector<int>& atoms, const robo::Real* atomMass) {
541 cartSolventAtoms_ = atoms;
542 cartSolventInvMass_.resize(atoms.size());
543 cartSolventMask_.assign(static_cast<std::size_t>(nAtoms_), 0);
544 for (std::size_t j = 0; j < atoms.size(); ++j) {
545 const int a = atoms[j];
546 cartSolventInvMass_[j] = (atomMass[a] > robo::Real(0)) ? (robo::Real(1) / atomMass[a]) : robo::Real(0);
547 if (a >= 0 && a < nAtoms_) {
548 cartSolventMask_[static_cast<std::size_t>(a)] = 1;
549 }
550 }
551 }
