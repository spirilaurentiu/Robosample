500 // primaryAxis aligned to primaryDir; planeAxis as close to planeVec as possible.
501 auto setRotationFromTwoAxes(const UnitVec3& primaryDir,
502 CoordinateAxis primaryAxis,
503 const Vec3& planeVec,
504 CoordinateAxis planeAxis) -> void {
505 const int idxU = static_cast<int>(primaryAxis);
506 const int idxV = static_cast<int>(planeAxis);
507 const int idxW = 3 - idxU - idxV;
508 const Vec3 colU = primaryDir.asVec3();
509 const Vec3 perp = planeVec - (colU * dot(planeVec, colU)); // Gram-Schmidt toward planeVec
510 const Real perpNorm = perp.norm();
511 const Vec3 colV = (perpNorm > Real(1e-12)) ? (perp / perpNorm) : anyUnitPerpendicular(colU);
512 const bool cyclic = (idxV == ((idxU + 1) % 3));
513 const Vec3 colW = cyclic ? (colU % colV) : (colV % colU);
514 setColumn(idxU, colU);
515 setColumn(idxV, colV);
516 setColumn(idxW, colW);
517 }

523 static auto fromQuaternion(const Quat& source) -> Rotation {
524 Quat normalized = source;
525 normalized.normalize();
526 const Real qw = normalized.elems[0];
527 const Real qx = normalized.elems[1];
528 const Real qy = normalized.elems[2];
529 const Real qz = normalized.elems[3];
530 Rotation rot;
531 rot.elems = {1 - (2 * ((qy * qy) + (qz * qz))),
532 2 * ((qx * qy) - (qw * qz)),
533 2 * ((qx * qz) + (qw * qy)),
534 2 * ((qx * qy) + (qw * qz)),
535 1 - (2 * ((qx * qx) + (qz * qz))),
536 2 * ((qy * qz) - (qw * qx)),
537 2 * ((qx * qz) - (qw * qy)),
538 2 * ((qy * qz) + (qw * qx)),
539 1 - (2 * ((qx * qx) + (qy * qy)))};
540 return rot;
541 }

563 const Real absX = std::abs(vec[0]);

589inline auto calcDihedralAngle(const Vec3& atomA, const Vec3& atomB, const Vec3& atomC, const Vec3& atomD)
590 -> Real {

593 const Vec3 edge3 = atomD - atomC;

741 pointMass * (-(posX * posY)),
742 pointMass * ((posX * posX) + (posZ * posZ)),
743 pointMass * (-(posX * posZ)),
744 pointMass * (-(posY * posZ)),
745 pointMass * ((posX * posX) + (posY * posY)));
746 }

795struct ArticulatedInertia {
796 SymMat33 angAng{SymMat33::zero()}; // J
797 Mat33 angLin{Mat33::zero()}; // F
798 SymMat33 linLin{SymMat33::zero()}; // M
799
800 ArticulatedInertia() = default;
801 ArticulatedInertia(const SymMat33& massBlock, const Mat33& momentBlock, const SymMat33& inertiaBlock)
802 : angAng(inertiaBlock)
803 , angLin(momentBlock)
804 , linLin(massBlock) {
805 }
806 explicit ArticulatedInertia(const SpatialInertia& spatial)
807 : angAng(spatial.unitInertia * spatial.mass)
808 , angLin(crossMat(spatial.com) * spatial.mass)
809 , linLin(SymMat33::diagonal(spatial.mass)) {
810 }
811
812 auto operator+=(const ArticulatedInertia& rhs) -> ArticulatedInertia& {
813 angAng += rhs.angAng;
814 angLin = angLin + rhs.angLin;
815 linLin += rhs.linLin;
816 return *this;
817 }
818 auto operator-(const ArticulatedInertia& rhs) const -> ArticulatedInertia {
819 ArticulatedInertia out;
820 out.angAng = angAng - rhs.angAng;
821 out.angLin = angLin - rhs.angLin;
822 out.linLin = linLin - rhs.linLin;
823 return out;
824 }
825 auto operator*(const SpatialVec& vec) const -> SpatialVec {
826 return {(angAng * vec.angular) + (angLin * vec.linear),
827 angLin.transposeTimes(vec.angular) + (linLin * vec.linear)};
828 }
829
830 // shift to a point `offset` away: Phi(offset) * P * ~Phi(offset).
831 [[nodiscard]] auto shift(const Vec3& offset) const -> ArticulatedInertia {
832 const Mat33 skew = crossMat(offset);
833 const Mat33 massFull = linLin.full();
834 const Mat33 momentFull = angLin;
835 const Mat33 newMoment = momentFull + (skew * massFull);
836 const Mat33 term1 = skew * momentFull.transpose(); // sx * ~F
837 const Mat33 term2 = momentFull * skew; // F * sx
838 const Mat33 term3 = (skew * massFull) * skew; // sx * M * sx
839 const Mat33 newInertiaFull = (angAng.full() + term1) - (term2 + term3);
840 ArticulatedInertia out;
841 out.linLin = linLin;
842 out.angLin = newMoment;
843 out.angAng = SymMat33::fromSymmetricPart(newInertiaFull);
844 return out;
845 }
846};
847
848inline auto SpatialInertia::operator*(const SpatialVec& vel) const -> SpatialVec {
849 return ArticulatedInertia(*this) * vel; // identical block math, single source of truth
850}
