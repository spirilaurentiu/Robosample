139 [[nodiscard]] auto getNumDegreesOfFreedom() const -> int {
140 ensureInitialized();
141 int dof = 0;
142 const int numParticles = system->getNumParticles();
143 for (int i = 0; i < numParticles; ++i) {
144 if (system->getParticleMass(i) > 0.0) {
145 dof += 3;
146 }
147 }
148 dof -= system->getNumConstraints();
149 const int numForces = system->getNumForces();
150 for (int i = 0; i < numForces; ++i) {
151 if (dynamic_cast<const OpenMM::CMMotionRemover*>(&system->getForce(i)) != nullptr) {
152 dof -= 3;
153 break;
154 }
155 }
