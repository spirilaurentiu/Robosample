42[[nodiscard]] inline double protocolLambda(int step, int ncmcSteps, double holdFraction) {
43 if (ncmcSteps <= 1) {
44 return 1.0; // degenerate: a single, fully-coupled (lambda = 1) substep
45 }
46 const double center = (ncmcSteps - 1) / 2.0; // palindrome axis (may be half-integer)
47 const double holdHalf = (holdFraction * ncmcSteps) / 2.0; // half-width of the flat lambda=0 hold
48 const double denom = center - holdHalf; // ramp half-width in index units
49 if (denom <= 0.0) {
50 return 0.0; // degenerate: the hold spans the whole protocol
51 }
52 // Symmetric tent: 1 at the endpoints (|step-center| == center), 0 inside the
53 // hold (|step-center| <= holdHalf), linear in between. Palindromic in step.
54 const double lam = (std::abs(static_cast<double>(step) - center) - holdHalf) / denom;
55 return std::clamp(lam, 0.0, 1.0);
56}
