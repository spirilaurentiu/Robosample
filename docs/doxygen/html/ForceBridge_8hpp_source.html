104 void getForcesFromOpenMM(RobotState& s) const {
105 std::vector<OpenMM::Vec3> forces;
106 OpenMMContext::get().evaluateForcesFromPositionsCache(posCache_, forces);
107
108 robo::SpatialVec* BF = s.bodyForceG();
109 for (int b = 0; b < model_.numBodies; ++b) {
110 BF[b] = robo::SpatialVec(robo::Vec3(0), robo::Vec3(0));
111 }
112 // Generalized (joint) forces. calcUDot reads mobilityForce as the
113 // applied per-DOF force (eps = f - H^T z). This model applies forces
114 // ONLY as Cartesian atom forces (reduced into bodyForceG above); there
115 // are no direct joint torques (Fixman unused), so the generalized force
116 // is identically zero. It lives in the arena and is never written
117 // elsewhere, so it MUST be cleared here every step -- otherwise
118 // calcUDot reads uninitialized slab memory -> garbage udot -> NaN.
119 // (Any future Fixman/biasing term should add into this cleared vector
120 // after evaluate() and before calcUDot.)
121 robo::Real* mob = s.mobilityForce();
122 for (int i = 0; i < model_.nu; ++i) {
123 mob[i] = robo::Real(0);
124 }
125 // When some atoms are Cartesian-integrated inside the proposal
126 // (solvent-relaxing NCMC), the velocity-Verlet for those atoms needs the
127 // raw per-atom Cartesian force. Cache it here (the only place the per-atom
128 // force vector exists) so the integrator can read it back. Skipped
129 // entirely otherwise -- the welded engine never allocates/reads it.
130 const bool cacheAtomForces = s.wantsAtomForces();
131 robo::Vec3* atomForce = cacheAtomForces ? s.atomForceG() : nullptr;
132 const robo::Vec3* posG = s.atomPosG();
133 const robo::Transform* X_GB = s.X_GB();
134
135 // Convert OpenMM's per-atom forces to robo::Vec3 (reduceAtomForcesToBodies
136 // is OpenMM-free) and cache them for the solvent Verlet where requested,
137 // under the SAME mass==0 skip the reduction below applies (a virtual site
138 // never needs its own Cartesian-Verlet force cached).
139 std::vector<robo::Vec3> forceG(static_cast<std::size_t>(model_.numAtoms));
140 for (int a = 0; a < model_.numAtoms; ++a) {
141 const robo::Vec3 f(forces[a][0], forces[a][1], forces[a][2]);
142 forceG[static_cast<std::size_t>(a)] = f;
143 if (atomForce && model_.atomMass[a] != robo::Real(0)) {
144 atomForce[a] = f; // per-atom Cartesian force for the solvent Verlet
145 }
146 }
147
148 // OpenMM -> Robosample body-wrench reduction (INV-1/INV-2), hoisted so the
149 // CUDA reduceForces kernel is pinned against this one definition
150 // (tests/TestForceReducer.cpp). Virtual sites (mass == 0, e.g. the
151 // OPC/TIP4P M-site) carry no independent DOF: OpenMM's getState(Forces)
152 // has ALREADY projected the force computed at the site onto its real
153 // parent atoms (verify: sum over REAL atoms reproduces the system net
154 // force / -dPE/dq to machine precision, while the raw force still left in
155 // the site's own slot makes the global sum nonzero). Reducing that
156 // leftover slot too would DOUBLE-COUNT the site force -- here it would
157 // re-apply ~10^3 kJ/mol/nm per water at the M-site station, a
158 // non-conservative kick that pumps the body's kinetic energy every step.
159 // reduceAtomForcesToBodies skips it; the parents already carry the
160 // contribution.
161 reduceAtomForcesToBodies(forceG.data(), posG, model_.atomMass.data(), model_.atomBody.data(), X_GB,
162 model_.numAtoms, model_.numBodies, BF);
163 }

235 OpenMMContext::get().setPositions(posCache_); // push to the live context
236 const bool ok =
237 OpenMMContext::get().integrateTrajectory(static_cast<int>(steps), static_cast<double>(timestep));

239 robo::Vec3* p = s.atomPosG();
240 for (int a = 0; a < model_.numAtoms; ++a) {

247 // Allocate the reused staging buffers once and build the constant virtual-site mask.
248 // stationFlat_ is only sized here; its CONTENTS are repacked every step (atomStation_B
249 // is refit per coordinate transfer, so it changes between rounds).
250 void ensureFusedStaging() {
251 if (fusedStagingBuilt_) {
252 return;
253 }
254 stationFlat_.assign(static_cast<std::size_t>(3 * model_.numAtoms), 0.0);
255 isVirtualI_.resize(static_cast<std::size_t>(model_.numAtoms));
256 for (int a = 0; a < model_.numAtoms; ++a) {

259 }
260 xgbFlat_.assign(static_cast<std::size_t>(12 * model_.numBodies), 0.0);
261 bodyForceFlat_.assign(static_cast<std::size_t>(6 * model_.numBodies), 0.0);
262 fusedStagingBuilt_ = true;

265 // Fused CUDA evaluate: push X_GB*station into OpenMM's device posq, evaluate forces +
266 // energy on device, and reduce forces to per-body spatial forces on device. Per robotics
267 // STEP the only host<->device traffic is X_GB up (12*numBodies) and bodyForceG down
268 // (6*numBodies); the atom stations (3*numAtoms) go up only when they change -- i.e. once
269 // per round, on a coordinate transfer (World::recomputeGeometry -> markStationsDirty).
270 void evaluateFused(RobotState& s) {
271 ensureFusedStaging();
272 OpenMMContext& omm = OpenMMContext::get();
273
274 // atomStation_B is refit only on a coordinate transfer (once per round), so repack
275 // the flat stations ONLY when they changed. Otherwise this per-atom pass is skipped.
276 if (stationsDirty_) {
277 for (int a = 0; a < model_.numAtoms; ++a) {
278 const robo::Vec3& st = model_.atomStation_B[static_cast<std::size_t>(a)];
279 stationFlat_[static_cast<std::size_t>(3 * a)] = st[0];
280 stationFlat_[static_cast<std::size_t>(3 * a + 1)] = st[1];
281 stationFlat_[static_cast<std::size_t>(3 * a + 2)] = st[2];
282 }
283 }
284 omm.ensureKinematicsConstants(static_cast<const void*>(this),
285 model_.numAtoms,
286 model_.numBodies,
287 stationFlat_.data(),
288 model_.atomBody.data(),
289 isVirtualI_.data(),
290 model_.bodyAtomsBeg.data(),
291 model_.bodyAtomsEnd.data(),
292 model_.bodyAtoms.data(),
293 /*stationsChanged=*/stationsDirty_);
294 stationsDirty_ = false;
295
296 // Pack X_GB as 12 doubles/body: 9 row-major rotation (robo::Mat33::elems) + 3
297 // translation. Matches the kernel's row-major R*station + p.
298 const robo::Transform* X_GB = s.X_GB();
299 for (int b = 0; b < model_.numBodies; ++b) {
300 const std::array<robo::Real, 9>& e = X_GB[b].R().elems;
301 double* d = xgbFlat_.data() + static_cast<std::size_t>(12 * b);
302 for (int k = 0; k < 9; ++k) {
303 d[k] = e[static_cast<std::size_t>(k)];
304 }
305 const robo::Vec3& p = X_GB[b].p();
306 d[9] = p[0];
307 d[10] = p[1];
308 d[11] = p[2];
309 }
310
311 omm.pushBodyTransforms(xgbFlat_.data()); // K1: posq <- X_GB*station
312 lastPE_ = omm.computeForcesAndEnergyOnDevice(); // forces+energy on device
313 lastEvalWasFused_ = true; // calcPotentialEnergy uses lastPE_
314 omm.reduceForcesToBodies(bodyForceFlat_.data()); // K2: forces -> per-body
315
316 robo::SpatialVec* BF = s.bodyForceG();
317 for (int b = 0; b < model_.numBodies; ++b) {
318 const double* f = bodyForceFlat_.data() + static_cast<std::size_t>(6 * b);
319 BF[b] = robo::SpatialVec(robo::Vec3(f[0], f[1], f[2]), // angular (moment about Bo)
320 robo::Vec3(f[3], f[4], f[5])); // linear (net force)
321 }
322 // Generalized (joint) forces are applied only as Cartesian atom forces, so the
323 // per-DOF force is identically zero -- clear it (calcUDot reads it). Mirrors the
324 // host getForcesFromOpenMM; any future Fixman/bias term adds here before calcUDot.
325 robo::Real* mob = s.mobilityForce();
326 for (int i = 0; i < model_.nu; ++i) {
327 mob[i] = robo::Real(0);
328 }
329 }
