38inline void rotationToQuaternion(const Rotation& R, Real& w, Real& x, Real& y, Real& z) {
39 const Real m00 = R(0, 0), m11 = R(1, 1), m22 = R(2, 2);
40 const Real tr = m00 + m11 + m22;
41 if (tr > 0) {
42 Real s = std::sqrt(tr + 1.0) * 2.0;
43 w = 0.25 * s;
44 x = (R(2, 1) - R(1, 2)) / s;
45 y = (R(0, 2) - R(2, 0)) / s;
46 z = (R(1, 0) - R(0, 1)) / s;
47 } else if (m00 > m11 && m00 > m22) {
48 Real s = std::sqrt(1.0 + m00 - m11 - m22) * 2.0;
49 w = (R(2, 1) - R(1, 2)) / s;
50 x = 0.25 * s;
51 y = (R(0, 1) + R(1, 0)) / s;
52 z = (R(0, 2) + R(2, 0)) / s;
53 } else if (m11 > m22) {
54 Real s = std::sqrt(1.0 + m11 - m00 - m22) * 2.0;
55 w = (R(0, 2) - R(2, 0)) / s;
56 x = (R(0, 1) + R(1, 0)) / s;
57 y = 0.25 * s;
58 z = (R(1, 2) + R(2, 1)) / s;
59 } else {
60 Real s = std::sqrt(1.0 + m22 - m00 - m11) * 2.0;
61 w = (R(1, 0) - R(0, 1)) / s;
62 x = (R(0, 2) + R(2, 0)) / s;
63 y = (R(1, 2) + R(2, 1)) / s;
64 z = 0.25 * s;
65 }
66}

76inline Real safeLogSineSqr(Real pitch) {
77 const Real s = std::sin(pitch);
78 Real s2 = s * s;
79 constexpr Real kFloor = 1e-12;
80 if (s2 < kFloor) {
81 s2 = kFloor;
82 }
83 return std::log(s2);
84}

95inline auto quatToRotation(Real qw, Real qx, Real qy, Real qz) -> Rotation {
96 const Real xx = qx * qx;
97 const Real yy = qy * qy;
98 const Real zz = qz * qz;
99 const Real xy = qx * qy;
100 const Real xz = qx * qz;
101 const Real yz = qy * qz;
102 const Real wx = qw * qx;
103 const Real wy = qw * qy;
104 const Real wz = qw * qz;
105 return {robo::Mat33(1 - (2 * (yy + zz)),
106 2 * (xy - wz),
107 2 * (xz + wy),
108 2 * (xy + wz),
109 1 - (2 * (xx + zz)),
110 2 * (yz - wx),
111 2 * (xz - wy),
112 2 * (yz + wx),
113 1 - (2 * (xx + yy)))};
114}
