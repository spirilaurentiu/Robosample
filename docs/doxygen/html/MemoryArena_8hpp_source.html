3// ============================================================================
4// MemoryArena — one aligned, contiguous memory slab with a bump pointer.
5//
6// Purpose: give each World (live RobotState) and each Replica (compact
7// RobotState) ONE cache-line-aligned block, internally partitioned into SoA
8// arrays. No per-array malloc, no vector growth, no pointer chasing between
9// arrays at run time: every array's offset is fixed at build time.
10//
11// OpenMP-future-compatible: every sub-array is 64-byte aligned, so later
12// `#pragma omp simd` / aligned AVX-512 loads are legal without re-layout.
13// We do NOT use OpenMP here.
14//
15// Usage:
16// MemoryArena arena;
17// // Phase 1: reserve (no allocation yet) — record offsets.
18// auto qOff = arena.reserve<double>(nq);
19// auto posOff = arena.reserve<double>(3 * nAtoms);
20// ...
21// arena.commit(); // single aligned allocation
22// double* q = arena.ptr<double>(qOff);
23// double* pos = arena.ptr<double>(posOff);
24// ============================================================================

120 void commit() {
121 if (committed_) {
122 return;
123 }

125 if (size_ == 0) {
126 size_ = kAlign;
127 }
